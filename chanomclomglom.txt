    const convertUnitToStock = (amount, recipeUnit, stockUnit) => {
      const rUnit = (recipeUnit || '').toLowerCase().trim();
      const sUnit = (stockUnit || '').toLowerCase().trim();

      if (rUnit === sUnit) return amount;
      if ((rUnit === 'กรัม' || rUnit === 'g') && (sUnit === 'กิโลกรัม' || sUnit === 'kg')) return amount / 1000;
      if ((rUnit === 'กิโลกรัม' || rUnit === 'kg') && (sUnit === 'กรัม' || sUnit === 'g')) return amount * 1000;
      if ((rUnit === 'มล.' || rUnit === 'ml') && (sUnit === 'ลิตร' || sUnit === 'l')) return amount / 1000;
      if ((rUnit === 'ลิตร' || rUnit === 'l') && (sUnit === 'มล.' || sUnit === 'ml')) return amount * 1000;

      return amount;
    };

    function EditProductionLogModal({ isOpen, onClose, logToEdit, onSave }) {
      const [itemsMap, setItemsMap] = useState({});
      const [note, setNote] = useState('');

      useEffect(() => {
        if (logToEdit) {
          setNote(logToEdit.note || '');
          const initialMap = {};
          // แกะ title ออกมา เช่น "ปรุง ชาไทย (2 ถัง), ชานม (1 ถัง)"
          if (logToEdit.title) {
            const raw = logToEdit.title.replace(/^(ปรุง|ต้ม)\s+/, '').split(', ');
            raw.forEach(item => {
              const match = item.match(/(.+) \((\d+) (ถัง|หม้อ)\)/);
              if (match) {
                const name = match[1].trim();
                const count = parseInt(match[2], 10);
                initialMap[name] = count;
              }
            });
          }
          setItemsMap(initialMap);
        }
      }, [logToEdit]);

      if (!isOpen || !logToEdit) return null;

      const handleQtyChange = (name, val) => {
        const num = Math.max(0, parseInt(val, 10) || 0);
        setItemsMap(prev => ({ ...prev, [name]: num }));
      };

      const handleRemoveItem = (name) => {
        setItemsMap(prev => {
          const updated = { ...prev };
          delete updated[name];
          return updated;
        });
      };

      const handleSubmit = (e) => {
        e.preventDefault();
        onSave(logToEdit, itemsMap, note);
      };

      return (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm animate-fade-in">
          <div className="bg-white rounded-3xl max-w-md w-full p-6 shadow-2xl border border-slate-100 max-h-[85vh] flex flex-col justify-between">

                </div>
                <button onClick={onClose} className="p-1 rounded-lg text-slate-400 hover:text-slate-600 hover:bg-slate-100 transition">
                  <i data-lucide="x" className="w-5 h-5"></i>
                </button>
              </div>

                  {Object.entries(itemsMap).map(([name, qty]) => (

                      <span className="font-bold text-xs text-slate-800 truncate flex-1">{name}</span>

                        <input
                          type="number"

    // 3. MODAL: เอาของออกไปหน้าร้าน (Transfer Batch Modal)
    // ==========================================
    function TransferModal({ isOpen, onClose, stocks, categories, onConfirmBatchTransfer }) {
      const [transferQtyMap, setTransferQtyMap] = useState({});
      const [note, setNote] = useState('');
      const [selectedCategory, setSelectedCategory] = useState('all');

      useEffect(() => {
        if (isOpen) {
          const initialMap = {};
          stocks.forEach(s => { initialMap[s.id] = ''; });
          setTransferQtyMap(initialMap);
          setNote('');
          setSelectedCategory('all');
        }
      }, [isOpen, stocks]);

      if (!isOpen) return null;

      const filteredStocks = stocks.filter(s => selectedCategory === 'all' || s.category === selectedCategory);

      const handleQtyChange = (stockId, val) => {
        setTransferQtyMap(prev => ({ ...prev, [stockId]: val }));
      };

      const handleSubmit = (e) => {
        e.preventDefault();

        for (const stockId in transferQtyMap) {
          const qty = Number(transferQtyMap[stockId]);
          if (qty > 0) {
            const stockItem = stocks.find(s => s.id === stockId);
            if (stockItem && qty > Number(stockItem.current)) {
              return alert(`รายการ "${stockItem.name}" เอาออกเกินจำนวนคงเหลือที่มีในคลัง (มีอยู่ ${stockItem.current} ${stockItem.unit})`);
            }
          }
        }

        onConfirmBatchTransfer(transferQtyMap, note);
      };

      return (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm animate-fade-in">
          <div className="bg-white rounded-3xl max-w-2xl w-full p-6 shadow-2xl border border-slate-100 max-h-[90vh] flex flex-col justify-between">

                  <i data-lucide="x" className="w-5 h-5"></i>
                </button>
              </div>

                <button
                  type="button"
                  onClick={() => setSelectedCategory('all')}
                  className={`px-3 py-1.5 rounded-xl text-xs font-bold transition-all whitespace-nowrap ${
                    selectedCategory === 'all'
                      ? 'bg-amber-500 text-white shadow-md shadow-amber-500/20'
                      : 'bg-slate-100 text-slate-600 hover:bg-slate-200'
                  }`}
                >
                  ทั้งหมด
                </button>
                {categories.map((cat) => (
                  <button
                    key={cat.id}
                    type="button"
                    onClick={() => setSelectedCategory(cat.name)}
                    className={`px-3 py-1.5 rounded-xl text-xs font-bold transition-all whitespace-nowrap ${
                      selectedCategory === cat.name
                        ? 'bg-amber-500 text-white shadow-md shadow-amber-500/20'
                        : 'bg-slate-100 text-slate-600 hover:bg-slate-200'

                              <span className="font-bold text-sm text-slate-800 truncate">{stock.name}</span>

              const name = match[1].trim();

    // ==========================================
    // 5. MODAL: เพิ่ม/แก้ไข สูตรเครื่องดื่ม/น้ำปรุง (BOM Modal)
    // ==========================================
    function RecipeModal({ isOpen, onClose, onSave, editingRecipe, stocks, handleImageUpload }) {
      const [name, setName] = useState('');
      const [yieldInfo, setYieldInfo] = useState('1 ถัง');
      const [image, setImage] = useState('');
      const [ingredients, setIngredients] = useState([]);

      useEffect(() => {
        if (editingRecipe) {
          setName(editingRecipe.name || '');
          setYieldInfo(editingRecipe.yieldInfo || '1 ถัง');
          setImage(editingRecipe.image || '');
          setIngredients(editingRecipe.ingredients || []);
        } else {
          setName('');
          setYieldInfo('1 ถัง');
          setImage('');
          setIngredients([]);
        }
      }, [editingRecipe, isOpen]);

      if (!isOpen) return null;

      const handleAddIngredient = () => {
        if (stocks.length === 0) return alert('กรุณาเพิ่มวัตถุดิบในคลังก่อนสร้างสูตร');
        const defaultStock = stocks[0];
        setIngredients(prev => [...prev, { stockId: defaultStock.id, amount: '', unit: defaultStock.unit || 'กรัม' }]);
      };

      const handleRemoveIngredient = (index) => {
        setIngredients(prev => prev.filter((_, i) => i !== index));
      };

      const handleIngredientChange = (index, field, value) => {
        setIngredients(prev => {
          const updated = [...prev];
          if (field === 'stockId') {
            const selectedStock = stocks.find(s => s.id === value);
            updated[index] = { 
              ...updated[index], 
              stockId: value, 
              unit: selectedStock ? selectedStock.unit : 'กรัม' 
            };
          } else {
            updated[index] = { ...updated[index], [field]: value };
          }
          return updated;
        });
      };

      const handleSubmit = (e) => {
        e.preventDefault();
        if (!name) return alert('กรุณาระบุชื่อสูตรน้ำปรุง');
        if (ingredients.length === 0) return alert('กรุณาเพิ่มส่วนผสมอย่างน้อย 1 รายการ');

        for (let ing of ingredients) {
          if (!ing.stockId || !ing.amount || Number(ing.amount) <= 0) {
            return alert('กรุณากรอกปริมาณส่วนผสมให้ครบถ้วนและถูกต้อง');
          }
        }

        onSave({
          id: editingRecipe ? editingRecipe.id : undefined,
          name,
          yieldInfo,
          image,
          ingredients: ingredients.map(i => ({ 
            stockId: i.stockId, 
            amount: Number(i.amount),
            unit: i.unit || 'กรัม'
          }))
        });
      };

      return (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm animate-fade-in">
          <div className="bg-white rounded-3xl max-w-lg w-full p-6 shadow-2xl border border-slate-100 max-h-[90vh] flex flex-col justify-between">
            <div className="overflow-y-auto pr-1">
              <div className="flex justify-between items-center pb-4 border-b border-slate-100 mb-5">
                <h3 className="font-bold text-lg text-slate-800 flex items-center gap-2">
                  <i data-lucide={editingRecipe ? "edit-3" : "book-open"} className="w-5 h-5 text-emerald-600"></i>
                  {editingRecipe ? 'แก้ไขสูตรน้ำปรุง' : 'เพิ่มสูตรน้ำปรุงใหม่ (BOM)'}
                </h3>
                <button onClick={onClose} className="p-1 rounded-lg text-slate-400 hover:text-slate-600 hover:bg-slate-100 transition">
                  <i data-lucide="x" className="w-5 h-5"></i>
                </button>
              </div>

              <form id="recipeForm" onSubmit={handleSubmit} className="space-y-4">
                <div>
                  <label className="block text-xs font-semibold text-slate-600 mb-1">ชื่อสูตร / น้ำปรุง <span className="text-rose-500">*</span></label>
                  <input type="text" placeholder="เช่น น้ำปรุงรสสูตร 1" value={name} onChange={e => setName(e.target.value)} required className="w-full px-4 py-3 text-sm rounded-xl custom-input" />
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-600 mb-1">ปริมาณที่ได้ต่อ 1 บัตช์ / ถัง</label>
                  <input type="text" placeholder="เช่น 1 ถัง" value={yieldInfo} onChange={e => setYieldInfo(e.target.value)} className="w-full px-4 py-2.5 text-sm rounded-xl custom-input" />
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-600 mb-1">รูปภาพประกอบ</label>
                  <div className="flex items-center space-x-3">
                    <img src={image || DEFAULT_IMAGE} className="w-12 h-12 rounded-xl object-cover border border-slate-200" alt="พรีวิวสูตร" />
                    <input type="file" accept="image/*" onChange={(e) => handleImageUpload(e, (base64) => setImage(base64))} className="text-xs text-slate-500 file:mr-3 file:py-2 file:px-4 file:rounded-xl file:border-0 file:text-xs file:font-semibold file:bg-emerald-50 file:text-emerald-700 hover:file:bg-emerald-100" />
                  </div>
                </div>

                <div className="pt-2 border-t border-slate-100">
                  <div className="flex justify-between items-center mb-3">
                    <label className="text-xs font-bold text-slate-700">รายการส่วนผสม (ต่อ 1 ถัง) <span className="text-rose-500">*</span></label>
                    <button type="button" onClick={handleAddIngredient} className="text-xs font-bold text-emerald-600 hover:text-emerald-700 bg-emerald-50 px-3 py-1.5 rounded-lg flex items-center gap-1">
                      <i data-lucide="plus" className="w-3.5 h-3.5"></i> เพิ่มวัตถุดิบ
                    </button>
                  </div>

                  {ingredients.length === 0 ? (
                    <div className="text-center py-6 border-2 border-dashed border-slate-200 rounded-2xl text-xs text-slate-400">
                      ยังไม่มีส่วนผสม กดปุ่ม "+ เพิ่มวัตถุดิบ" ด้านบนเพื่อระบุส่วนผสม
                    </div>
                  ) : (
                    <div className="space-y-3">
                      {ingredients.map((ing, idx) => (
                        <div key={idx} className="flex items-center space-x-2 bg-slate-50 p-2.5 rounded-xl border border-slate-200">
                          <select
                            value={ing.stockId}
                            onChange={(e) => handleIngredientChange(idx, 'stockId', e.target.value)}
                            className="flex-1 px-2.5 py-2 text-xs rounded-lg custom-input font-medium"
                          >
                            {stocks.map(s => (
                              <option key={s.id} value={s.id}>{s.name}</option>
                            ))}
                          </select>

                          <input
                            type="number"
                            step="any"
                            placeholder="จำนวน"
                            value={ing.amount}
                            onChange={(e) => handleIngredientChange(idx, 'amount', e.target.value)}
                            className="w-20 px-2 py-2 text-xs rounded-lg custom-input font-bold text-center"
                          />

                          <input
                            type="text"
                            placeholder="หน่วย"
                            value={ing.unit || ''}
                            onChange={(e) => handleIngredientChange(idx, 'unit', e.target.value)}
                            className="w-16 px-2 py-2 text-xs rounded-lg custom-input text-center"
                          />

                          <button type="button" onClick={() => handleRemoveIngredient(idx)} className="p-1.5 text-slate-400 hover:text-rose-600 rounded-lg">
                            <i data-lucide="trash-2" className="w-4 h-4"></i>
                          </button>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              </form>
            </div>

            <div className="pt-4 border-t border-slate-100 flex gap-3 mt-4">
              <button type="button" onClick={onClose} className="flex-1 py-3 bg-slate-100 hover:bg-slate-200 text-slate-600 font-bold text-sm rounded-xl transition">ยกเลิก</button>
              <button type="submit" form="recipeForm" className="flex-1 py-3 bg-emerald-500 hover:bg-emerald-600 text-white font-bold text-sm rounded-xl shadow-lg shadow-emerald-500/20 transition">บันทึกสูตร</button>
            </div>
          </div>
        </div>
      );
    }

        );
      }, [logs, searchHistory]);

      // ดึงรายการเฉพาะ วัตถุดิบเข้า-ออก (เติมสต๊อก / เอาของออก)
      const filteredMovements = useMemo(() => {
        const movementLogs = logs.filter(l => l.type === 'เติมสต๊อก' || l.type === 'เอาของออกไปหน้าร้าน' || l.type === 'ย้ายไปหน้าร้าน');
        return movementLogs.filter(l =>
          (l.title && l.title.toLowerCase().includes(searchMovement.toLowerCase())) ||
          (l.datetime && l.datetime.toLowerCase().includes(searchMovement.toLowerCase())) ||
          (l.note && l.note.toLowerCase().includes(searchMovement.toLowerCase())) ||
          (Array.isArray(l.details) && l.details.some(d => d.toLowerCase().includes(searchMovement.toLowerCase())))
        );
      }, [logs, searchMovement]);

      const lowStockItems = useMemo(() => {
        return stocks.filter(item => {
          const currentVal = Number(item.current) || 0;
          const minVal = Number(item.minAlert) || 0;
          return currentVal <= minVal;
        });
      }, [stocks]);

      const handleImageUpload = (e, callback) => {
        const file = e.target.files[0];
        if (file) {
          if (file.size > 1.2 * 1024 * 1024) {
            alert("รูปภาพมีขนาดใหญ่เกินไป กรุณาใช้รูปขนาดไม่เกิน 1.2MB ครับ");
            return;
          }
          const reader = new FileReader();
          reader.onloadend = () => callback(reader.result);
          reader.readAsDataURL(file);
        }
      };

      // --- Production Handlers ---
      const handleQuantityChange = (recipeId, delta) => {
        setBatchQuantities(prev => {
          const current = prev[recipeId] || 0;
          const updated = Math.max(0, current + delta);
          return { ...prev, [recipeId]: updated };
        });
      };

      const handleRemoveFromProductionList = (recipeId) => {
        setBatchQuantities(prev => {
          const updated = { ...prev };
          delete updated[recipeId];
          return updated;
        });
      };

      const handleSetQuantityDirectly = (recipeId, val) => {
        const num = Math.max(0, parseInt(val, 10) || 0);
        setBatchQuantities(prev => ({ ...prev, [recipeId]: num }));
      };

      const selectedProductionList = useMemo(() => {
        return Object.entries(batchQuantities)
          .filter(([_, qty]) => qty > 0)
          .map(([recipeId, qty]) => {
            const recipe = recipes.find(r => r.id === recipeId);
            return {
              recipeId,
              name: recipe ? recipe.name : 'สูตรน้ำปรุง',
              qty,
              yieldInfo: recipe ? recipe.yieldInfo : '',
              recipe
            };
          });
      }, [batchQuantities, recipes]);

      // จำนวนถังรวมทั้งหมดที่ถูกเลือก
      const totalBucketsCount = useMemo(() => {
        return selectedProductionList.reduce((acc, curr) => acc + curr.qty, 0);
      }, [selectedProductionList]);

      const handleConfirmProduction = async () => {
        if (selectedProductionList.length === 0) return alert('กรุณาเลือกจำนวนถังอย่างน้อย 1 รายการ');

        try {
          const batch = db.batch();
          const deductionsSummary = {};

          selectedProductionList.forEach(item => {
            if (item.recipe && item.recipe.ingredients) {
              item.recipe.ingredients.forEach(ing => {
                const stockItem = stocks.find(s => s.id === ing.stockId);
                if (stockItem) {
                  const rawAmount = Number(ing.amount) * item.qty;
                  const convertedAmount = convertUnitToStock(rawAmount, ing.unit, stockItem.unit);
                  deductionsSummary[ing.stockId] = (deductionsSummary[ing.stockId] || 0) + convertedAmount;
                }
              });
            }
          });

          Object.entries(deductionsSummary).forEach(([stockId, deductAmount]) => {
            const stockRef = db.collection('stocks').doc(stockId);
            batch.update(stockRef, {
              current: firebase.firestore.FieldValue.increment(-deductAmount)
            });
          });

          const titleText = selectedProductionList.map(item => `${item.name} (${item.qty} ถัง)`).join(', ');
          
          const detailsList = Object.entries(deductionsSummary).map(([stockId, deductAmount]) => {
            const stockItem = stocks.find(s => s.id === stockId);
            return `${stockItem ? stockItem.name : 'วัตถุดิบ'}: -${deductAmount.toFixed(2)} ${stockItem ? stockItem.unit : ''}`;
          });

          const now = new Date();
          const day = now.getDate().toString().padStart(2, '0');
          const month = (now.getMonth() + 1).toString().padStart(2, '0');
          const year = now.getFullYear() + 543;
          const hours = now.getHours().toString().padStart(2, '0');
          const mins = now.getMinutes().toString().padStart(2, '0');
          const timeStr = `${day}/${month}/${year} ${hours}:${mins} น.`;

          const logRef = db.collection('logs').doc();
          batch.set(logRef, {
            datetime: timeStr,
            timestamp: firebase.firestore.FieldValue.serverTimestamp(),
            type: 'ปรุงผลิต',
            title: `ปรุง ${titleText}`,
            details: detailsList,
            rawDeductions: deductionsSummary,
            note: productionNote || 'ปรุงผลิตประจำวัน'
          });

          await batch.commit();
          alert('บันทึกการผลิตและตัดสต๊อกเรียบร้อยแล้ว!');
          setBatchQuantities({});
          setProductionNote('');
        } catch (error) {
          console.error("Error confirming production:", error);
          alert("เกิดข้อผิดพลาดในการบันทึกข้อมูล");
        }
      };

      // --- Movement Edit Handler ---
      const handleSaveMovementEdit = async (logItem, updatedItemsMap, newNote) => {
        try {
          const batch = db.batch();
          const isRestock = logItem.type === 'เติมสต๊อก';
          const oldMap = isRestock ? logItem.rawRestock : logItem.rawDeductions;

          if (oldMap) {
            // คืนค่าสต๊อกเดิมก่อน
            Object.entries(oldMap).forEach(([sId, oldQty]) => {
              const stockRef = db.collection('stocks').doc(sId);
              batch.update(stockRef, {
                current: firebase.firestore.FieldValue.increment(isRestock ? -Number(oldQty) : Number(oldQty))
              });
            });
          }

          // ปรับสต๊อกตามค่าใหม่
          const newDetailsList = [];
          const newRaw = {};

          Object.entries(updatedItemsMap).forEach(([sId, newQtyStr]) => {
            const newQty = Number(newQtyStr) || 0;
            if (newQty > 0) {
              const stockRef = db.collection('stocks').doc(sId);
              const stockItem = stocks.find(s => s.id === sId);
              batch.update(stockRef, {
                current: firebase.firestore.FieldValue.increment(isRestock ? newQty : -newQty)
              });
              newDetailsList.push(`${stockItem ? stockItem.name : 'วัตถุดิบ'}: ${isRestock ? '+' : '-'}${newQty} ${stockItem ? stockItem.unit : ''}`);
              newRaw[sId] = newQty;
            }
          });

          const logRef = db.collection('logs').doc(logItem.id);
          const updatePayload = {
            details: newDetailsList,
            note: newNote,
            title: isRestock ? `เติมวัตถุดิบเข้าคลัง (${Object.keys(newRaw).length} รายการ)` : `เอาของออกไปหน้าร้าน (${Object.keys(newRaw).length} รายการ)`
          };

          if (isRestock) updatePayload.rawRestock = newRaw;
          else updatePayload.rawDeductions = newRaw;

          batch.update(logRef, updatePayload);

          await batch.commit();
          setMovementToEdit(null);
          alert('แก้ไขบันทึกวัตถุดิบและปรับยอดสต๊อกเรียบร้อยแล้ว!');
        } catch (e) {
          console.error(e);
          alert('เกิดข้อผิดพลาดในการแก้ไขข้อมูล');
        }
      };

      // --- Save Production Edit (ลบชาไทยออก/ปรับถัง) ---
      const handleSaveProductionEdit = async (logItem, updatedItemsMap, newNote) => {
        try {
          const titleText = Object.entries(updatedItemsMap)
            .map(([name, qty]) => `${name} (${qty} ถัง)`)
            .join(', ');

          const logRef = db.collection('logs').doc(logItem.id);
          await logRef.update({
            title: titleText ? `ปรุง ${titleText}` : 'ยกเลิกการปรุงผลิต',
            note: newNote
          });

          setProductionToEdit(null);
          alert('แก้ไขบันทึกการผลิตเรียบร้อยแล้ว!');
        } catch (e) {
          console.error(e);
          alert('เกิดข้อผิดพลาดในการแก้ไข');
        }
      };

      const handleConfirmDeleteLog = async (shouldRestore) => {
        if (!logToDelete) return;

        try {
          const batch = db.batch();

          if (shouldRestore) {
            if (logToDelete.type === 'เติมสต๊อก') {
              if (logToDelete.rawRestock) {
                Object.entries(logToDelete.rawRestock).forEach(([sId, amount]) => {
                  const stockRef = db.collection('stocks').doc(sId);
                  batch.update(stockRef, {
                    current: firebase.firestore.FieldValue.increment(-Number(amount))
                  });
                });
              }
            } else {
              if (logToDelete.rawDeductions) {
                Object.entries(logToDelete.rawDeductions).forEach(([sId, amount]) => {
                  const stockRef = db.collection('stocks').doc(sId);
                  batch.update(stockRef, {
                    current: firebase.firestore.FieldValue.increment(Number(amount))
                  });
                });
              }
            }
          }

          const logRef = db.collection('logs').doc(logToDelete.id);
          batch.delete(logRef);

          await batch.commit();
          alert(shouldRestore ? 'ลบประวัติพร้อมคืนค่าสต๊อกเดิมเรียบร้อยแล้ว!' : 'ลบรายการประวัติเรียบร้อยแล้ว!');
          setLogToDelete(null);
        } catch (e) {
          console.error(e);
          alert('เกิดข้อผิดพลาดในการลบรายการประวัติ');
        }
      };

      // --- Stock Handlers ---
      const handleDeleteStock = async (id) => {
        if (confirm('คุณต้องการลบวัตถุดิบรายการนี้ใช่หรือไม่?')) {
          await db.collection('stocks').doc(id).delete();
        }
      };

      const handleSaveStock = async (stockData) => {
        try {
          if (stockData.id) {
            const id = stockData.id;
            delete stockData.id;
            await db.collection('stocks').doc(id).update(stockData);
          } else {
            delete stockData.id;
            await db.collection('stocks').add(stockData);
          }
          setStockModalOpen(false);
          alert('บันทึกข้อมูลวัตถุดิบเรียบร้อยแล้ว');
        } catch (err) {
          console.error("Error saving stock:", err);
          alert("ไม่สามารถบันทึกวัตถุดิบได้ กรุณาลองใหม่อีกครั้ง");
        }
      };

      // --- Restock Handlers ---
      const handleBatchRestock = async (restockItemsMap, globalNote) => {
        try {
          const itemsToProcess = Object.entries(restockItemsMap).filter(([_, qty]) => Number(qty) > 0);
          if (itemsToProcess.length === 0) return alert('กรุณาระบุจำนวนวัตถุดิบที่ต้องการเติมอย่างน้อย 1 รายการ');

          const batch = db.batch();
          const detailsList = [];
          const rawRestock = {};

          itemsToProcess.forEach(([stockId, qty]) => {
            const stockItem = stocks.find(s => s.id === stockId);
            const amount = Number(qty);
            if (stockItem && amount > 0) {
              const stockRef = db.collection('stocks').doc(stockId);
              batch.update(stockRef, {
                current: firebase.firestore.FieldValue.increment(amount)
              });
              detailsList.push(`${stockItem.name}: +${amount} ${stockItem.unit}`);
              rawRestock[stockId] = amount;
            }
          });

          const now = new Date();
          const day = now.getDate().toString().padStart(2, '0');
          const month = (now.getMonth() + 1).toString().padStart(2, '0');
          const year = now.getFullYear() + 543;
          const hours = now.getHours().toString().padStart(2, '0');
          const mins = now.getMinutes().toString().padStart(2, '0');
          const timeStr = `${day}/${month}/${year} ${hours}:${mins} น.`;

          const logRef = db.collection('logs').doc();
          batch.set(logRef, {
            datetime: timeStr,
            timestamp: firebase.firestore.FieldValue.serverTimestamp(),
            type: 'เติมสต๊อก',
            title: `เติมวัตถุดิบเข้าคลัง (${itemsToProcess.length} รายการ)`,
            details: detailsList,
            rawRestock: rawRestock,
            note: globalNote || 'เติมวัตถุดิบประจำวัน'
          });

          await batch.commit();
          setRestockModalOpen(false);
          alert(`เติมสต๊อกเรียบร้อยแล้วทั้งหมด ${itemsToProcess.length} รายการ!`);
        } catch (err) {
          console.error("Batch Restock error:", err);
          alert("เกิดข้อผิดพลาดในการเติมสต๊อก");
        }
      };

      // --- Transfer Batch Handler ---
      const handleBatchTransfer = async (transferItemsMap, globalNote) => {
        try {
          const itemsToProcess = Object.entries(transferItemsMap).filter(([_, qty]) => Number(qty) > 0);
          if (itemsToProcess.length === 0) return alert('กรุณาระบุจำนวนวัตถุดิบที่ต้องการเอาออกอย่างน้อย 1 รายการ');

          const batch = db.batch();
          const detailsList = [];
          const rawDeductions = {};

          itemsToProcess.forEach(([stockId, qty]) => {
            const stockItem = stocks.find(s => s.id === stockId);
            const amount = Number(qty);
            if (stockItem && amount > 0) {
              const stockRef = db.collection('stocks').doc(stockId);
              batch.update(stockRef, {
                current: firebase.firestore.FieldValue.increment(-amount)
              });
              detailsList.push(`${stockItem.name}: -${amount} ${stockItem.unit}`);
              rawDeductions[stockId] = amount;
            }
          });

          const now = new Date();
          const day = now.getDate().toString().padStart(2, '0');
          const month = (now.getMonth() + 1).toString().padStart(2, '0');
          const year = now.getFullYear() + 543;
          const hours = now.getHours().toString().padStart(2, '0');
          const mins = now.getMinutes().toString().padStart(2, '0');
          const timeStr = `${day}/${month}/${year} ${hours}:${mins} น.`;

          const logRef = db.collection('logs').doc();
          batch.set(logRef, {
            datetime: timeStr,
            timestamp: firebase.firestore.FieldValue.serverTimestamp(),
            type: 'เอาของออกไปหน้าร้าน',
            title: `เอาของออกไปหน้าร้าน (${itemsToProcess.length} รายการ)`,
            details: detailsList,
            rawDeductions: rawDeductions,
            note: globalNote || 'เอาของออกไปหน้าร้าน'
          });

          await batch.commit();
          setTransferModalOpen(false);
          alert(`เอาของออกไปหน้าร้านเรียบร้อยแล้วทั้งหมด ${itemsToProcess.length} รายการ!`);
        } catch (err) {
          console.error("Batch Transfer error:", err);
          alert("เกิดข้อผิดพลาดในการเอาของออกไปหน้าร้าน");
        }
      };

      // --- BOM Handlers ---
      const handleDeleteRecipe = async (id) => {
        if (confirm('คุณต้องการลบสูตรวัตถุดิบนี้ใช่หรือไม่?')) {
          await db.collection('recipes').doc(id).delete();
        }
      };

      const handleSaveRecipe = async (recipeObj) => {
        try {
          if (recipeObj.id) {
            const id = recipeObj.id;
            delete recipeObj.id;
            await db.collection('recipes').doc(id).update(recipeObj);
          } else {
            delete recipeObj.id;
            await db.collection('recipes').add(recipeObj);
          }
          setRecipeModalOpen(false);
          alert('บันทึกสูตรเรียบร้อยแล้ว');
        } catch (err) {
          console.error("Error saving recipe:", err);
          alert("ไม่สามารถบันทึกสูตรได้ กรุณาลองใหม่อีกครั้ง");
        }
      };

      return (
        <div className="min-h-screen flex flex-col justify-between">
          <div>
            {/* Header */}
            <header className="bg-white border-b border-slate-200 sticky top-0 z-40 shadow-sm">
              <div className="max-w-7xl mx-auto px-4 h-16 flex items-center justify-between border-b border-slate-100 sm:border-none">
                <div className="flex items-center space-x-2.5">
                  <div className="bg-emerald-500 text-white p-2 rounded-xl shadow-md shadow-emerald-500/20">
                    <i data-lucide="soup" className="w-5 h-5 sm:w-6 sm:h-6"></i>
                  </div>
                  <div>
                    <h1 className="font-bold text-slate-800 text-base sm:text-lg tracking-tight leading-none">ร้านติดใจ</h1>
                    <p className="text-[10px] text-slate-400 font-medium mt-0.5">ระบบจัดการวัตถุดิบ & สต๊อกน้ำปรุง</p>
                  </div>
                </div>

                <div className="flex items-center space-x-1.5 sm:space-x-2">
                  <button onClick={() => setPrintModalOpen(true)} className="bg-slate-900 text-white hover:bg-slate-800 text-xs font-bold px-3 py-2 rounded-xl flex items-center gap-1.5 shadow-md transition">
                    <i data-lucide="printer" className="w-3.5 h-3.5 sm:w-4 sm:h-4"></i>
                    <span className="hidden sm:inline">สรุปรายงานประจำวัน</span>
                  </button>

                  {lowStockItems.length > 0 && (
                    <button onClick={() => handleTabChange('stock')} className="bg-rose-50 text-rose-600 border border-rose-200 text-xs font-bold px-2.5 py-2 rounded-xl flex items-center gap-1 animate-pulse">
                      <i data-lucide="bell" className="w-3.5 h-3.5"></i>
                      <span>({lowStockItems.length})</span>
                    </button>
                  )}
                </div>
              </div>

              <div className="bg-slate-100/80 p-1.5 border-t border-slate-200/60">
                <div className="max-w-7xl mx-auto flex space-x-1 overflow-x-auto no-scrollbar justify-start sm:justify-center px-2">
                  <button onClick={() => handleTabChange('production')} className={`flex items-center space-x-1.5 px-3.5 py-2.5 rounded-xl text-xs font-bold transition-all whitespace-nowrap ${activeTab === 'production' ? 'bg-white text-emerald-600 shadow-sm border border-slate-200' : 'text-slate-600 hover:text-slate-900'}`}>
                    <i data-lucide="flame" className="w-4 h-4"></i>
                    <span>สั่งปรุงน้ำปรุง</span>
                  </button>

                  <button onClick={() => handleTabChange('stock')} className={`flex items-center space-x-1.5 px-3.5 py-2.5 rounded-xl text-xs font-bold transition-all whitespace-nowrap relative ${activeTab === 'stock' ? 'bg-white text-emerald-600 shadow-sm border border-slate-200' : 'text-slate-600 hover:text-slate-900'}`}>
                    <i data-lucide="boxes" className="w-4 h-4"></i>
                    <span>คลังวัตถุดิบ</span>
                    {lowStockItems.length > 0 && <span className="w-2 h-2 rounded-full bg-rose-500"></span>}
                  </button>

                  <button onClick={() => handleTabChange('bom')} className={`flex items-center space-x-1.5 px-3.5 py-2.5 rounded-xl text-xs font-bold transition-all whitespace-nowrap ${activeTab === 'bom' ? 'bg-white text-emerald-600 shadow-sm border border-slate-200' : 'text-slate-600 hover:text-slate-900'}`}>
                    <i data-lucide="book-open" className="w-4 h-4"></i>
                    <span>สูตรการผลิต (BOM)</span>
                    {!isBomUnlocked && <span className="text-[10px]">🔒</span>}
                  </button>

                  <button onClick={() => handleTabChange('history')} className={`flex items-center space-x-1.5 px-3.5 py-2.5 rounded-xl text-xs font-bold transition-all whitespace-nowrap ${activeTab === 'history' ? 'bg-white text-emerald-600 shadow-sm border border-slate-200' : 'text-slate-600 hover:text-slate-900'}`}>
                    <i data-lucide="history" className="w-4 h-4"></i>
                    <span>บันทึกการผลิต</span>
                  </button>

                  <button onClick={() => handleTabChange('movements')} className={`flex items-center space-x-1.5 px-3.5 py-2.5 rounded-xl text-xs font-bold transition-all whitespace-nowrap ${activeTab === 'movements' ? 'bg-white text-emerald-600 shadow-sm border border-slate-200' : 'text-slate-600 hover:text-slate-900'}`}>
                    <i data-lucide="arrow-left-right" className="w-4 h-4"></i>
                    <span>บันทึกวัตถุดิบ เข้า - ออก</span>
                  </button>
                </div>
              </div>
            </header>

            {/* Main Content Body */}
            <main className="max-w-7xl mx-auto px-3 sm:px-6 lg:px-8 my-4 sm:my-6">

              {loading ? (
                <div className="text-center py-20 text-slate-400 font-medium text-sm">กำลังโหลดข้อมูลระบบ...</div>
              ) : (
                <>
                  {/* TAB 1: สั่งปรุงน้ำปรุง */}
                  {activeTab === 'production' && (
                    <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
                      <div className="lg:col-span-7 bg-white rounded-3xl p-4 sm:p-8 border border-slate-200/80 custom-shadow">
                        <div className="mb-4 sm:mb-6">
                          <h2 className="text-lg sm:text-xl font-bold text-slate-800 flex items-center gap-2">
                            <span>📝</span> บันทึกปรุงน้ำรสประจำวัน
                          </h2>
                          <p className="text-xs text-slate-400 mt-1">กดเพิ่ม/ลดจำนวนถังที่การ์ดเมนู เพื่อสรุปรายการปรุงอัตโนมัติ</p>
                        </div>

                        <div className="mb-4 relative">
                          <input
                            type="text"
                            placeholder="ค้นหาชื่อสูตร / เมนูน้ำปรุง..."
                            value={searchProduction}
                            onChange={(e) => setSearchProduction(e.target.value)}
                            className="w-full px-4 py-3 text-xs sm:text-sm rounded-xl custom-input pl-10"
                          />

                          <span>📜</span> ประวัติบันทึกการผลิต

                        <p className="text-xs text-slate-400 mt-1">ตรวจสอบประวัติการปรุงน้ำรส ยอดตัดสต๊อก หรือแก้ไข/ลบรายการที่บันทึกผิด</p>
                      </div>

                      <div className="mb-5 relative">
                        <input
                          type="text"
                          placeholder="ค้นหาประวัติการผลิต..."
                          value={searchHistory}
                          onChange={(e) => setSearchHistory(e.target.value)}
                          className="w-full px-4 py-3 text-xs sm:text-sm rounded-xl custom-input pl-10"
                        />
                        <span className="absolute left-3.5 top-3.5 text-slate-400 text-xs">🔍</span>
                      </div>

                      {filteredLogs.length === 0 ? (
                        <div className="text-center py-16 text-slate-400 text-xs sm:text-sm">ยังไม่มีประวัติการสั่งปรุง</div>
                      ) : (
                        <div className="space-y-3">
                          {filteredLogs.map(log => {
                            const isShowDetails = unlockedLogDetails[log.id];
                            return (
                              <div key={log.id} className="bg-slate-50 border border-slate-200 p-4 rounded-2xl flex flex-col justify-between gap-3">
                                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 border-b border-slate-200/60 pb-2.5">
                                  <div>
                                    <div className="flex items-center gap-2">
                                      <span className="bg-emerald-100 text-emerald-700 text-[10px] font-bold px-2 py-0.5 rounded-md">
                                        {log.type || 'ปรุงผลิต'}
                                      </span>
                                      <span className="text-[11px] text-slate-400 font-medium">{log.datetime}</span>
                                    </div>
                                    <h3 className="font-bold text-sm sm:text-base text-slate-800 mt-1">{log.title}</h3>
                                    {log.note && <p className="text-xs text-slate-500 italic mt-0.5">หมายเหตุ: {log.note}</p>}
                                  </div>

                                  <div className="flex items-center gap-2 self-end sm:self-center">
                                    <button
                                      onClick={() => setProductionToEdit(log)}
                                      className="text-xs font-bold text-amber-600 bg-amber-50 hover:bg-amber-100 border border-amber-200 px-3 py-1.5 rounded-xl transition flex items-center gap-1"
                                    >
                                      <i data-lucide="edit-3" className="w-3.5 h-3.5"></i> แก้ไข
                                    </button>
                                    <button
                                      onClick={() => setLogToDelete(log)}
                                      className="text-xs font-bold text-rose-600 bg-rose-50 hover:bg-rose-100 border border-rose-200 px-3 py-1.5 rounded-xl transition flex items-center gap-1"
                                    >
                                      <i data-lucide="trash-2" className="w-3.5 h-3.5"></i> ลบ
                                    </button>
                                  </div>
                                </div>

                                <div>
                                  <div className="flex justify-between items-center">
                                    <button
                                      onClick={() => handleToggleShowDetails(log.id)}
                                      className="text-xs font-bold text-indigo-600 hover:underline flex items-center gap-1"
                                    >
                                      <i data-lucide={isShowDetails ? "eye-off" : "eye"} className="w-3.5 h-3.5"></i>
                                      {isShowDetails ? 'ซ่อนส่วนผสมที่ถูกตัด' : '🔒 ใส่รหัสเพื่อดูวัตถุดิบที่ตัดสต๊อก'}
                                    </button>
                                  </div>

                                  {isShowDetails && log.details && (
                                    <div className="mt-2.5 pt-2.5 border-t border-slate-200 grid grid-cols-2 sm:grid-cols-3 gap-1.5">
                                      {log.details.map((d, idx) => (
                                        <div key={idx} className="bg-white px-2.5 py-1.5 rounded-lg border border-slate-200 text-[11px] font-medium text-slate-700">
                                          {d}
                                        </div>
                                      ))}
                                    </div>
                                  )}
                                </div>
                              </div>
                            );
                          })}
                        </div>
                      )}
                    </div>
                  )}

                  {/* TAB 5: บันทึกวัตถุดิบ เข้า - ออก */}
                  {activeTab === 'movements' && (
                    <div className="bg-white rounded-3xl p-4 sm:p-8 border border-slate-200/80 custom-shadow">
                      <div className="mb-6 pb-4 border-b border-slate-100">
                        <h2 className="text-lg sm:text-xl font-bold text-slate-800 flex items-center gap-2">
                          <span>🔄</span> บันทึกการเข้า-ออกของวัตถุดิบ
                        </h2>
                        <p className="text-xs text-slate-400 mt-1">ดูประวัติการเติมของเข้าคลัง หรือการเบิกเอาของออกไปหน้าร้าน</p>
                      </div>

                      <div className="mb-5 relative">
                        <input
                          type="text"
                          placeholder="ค้นหาบันทึกการเบิก/เติมวัตถุดิบ..."
                          value={searchMovement}
                          onChange={(e) => setSearchMovement(e.target.value)}
                          className="w-full px-4 py-3 text-xs sm:text-sm rounded-xl custom-input pl-10"
                        />
                        <span className="absolute left-3.5 top-3.5 text-slate-400 text-xs">🔍</span>
                      </div>

                      {filteredMovements.length === 0 ? (
                        <div className="text-center py-16 text-slate-400 text-xs sm:text-sm">ยังไม่มีประวัติการเติมหรือเบิกวัตถุดิบ</div>
                      ) : (
                        <div className="space-y-3">
                          {filteredMovements.map(log => {
                            const isRestock = log.type === 'เติมสต๊อก';
                            return (
                              <div key={log.id} className="bg-slate-50 border border-slate-200 p-4 rounded-2xl flex flex-col justify-between gap-3">
                                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 border-b border-slate-200/60 pb-2.5">
                                  <div>
                                    <div className="flex items-center gap-2">
                                      <span className={`text-[10px] font-bold px-2 py-0.5 rounded-md ${isRestock ? 'bg-emerald-100 text-emerald-700' : 'bg-amber-100 text-amber-700'}`}>
                                        {log.type}
                                      </span>
                                      <span className="text-[11px] text-slate-400 font-medium">{log.datetime}</span>
                                    </div>
                                    <h3 className="font-bold text-sm sm:text-base text-slate-800 mt-1">{log.title}</h3>
                                    {log.note && <p className="text-xs text-slate-500 italic mt-0.5">หมายเหตุ: {log.note}</p>}
                                  </div>

                                  <div className="flex items-center gap-2 self-end sm:self-center">
                                    <button
                                      onClick={() => setMovementToEdit(log)}
                                      className="text-xs font-bold text-amber-600 bg-amber-50 hover:bg-amber-100 border border-amber-200 px-3 py-1.5 rounded-xl transition flex items-center gap-1"
                                    >
                                      <i data-lucide="edit-3" className="w-3.5 h-3.5"></i> แก้ไข
                                    </button>
                                    <button
                                      onClick={() => setLogToDelete(log)}
                                      className="text-xs font-bold text-rose-600 bg-rose-50 hover:bg-rose-100 border border-rose-200 px-3 py-1.5 rounded-xl transition flex items-center gap-1"
                                    >
                                      <i data-lucide="trash-2" className="w-3.5 h-3.5"></i> ลบ
                                    </button>
                                  </div>
                                </div>

                                {log.details && (
                                  <div className="grid grid-cols-2 sm:grid-cols-3 gap-1.5">
                                    {log.details.map((d, idx) => (
                                      <div key={idx} className="bg-white px-2.5 py-1.5 rounded-lg border border-slate-200 text-[11px] font-medium text-slate-700">
                                        {d}
                                      </div>
                                    ))}
                                  </div>
                                )}
                              </div>
                            );
                          })}
                        </div>
                      )}
                    </div>
                  )}
                </>
              )}
            </main>
          </div>

          {/* Footer */}
          <footer className="bg-white border-t border-slate-200 py-4 text-center text-xs text-slate-400 mt-8">
            <p>© {new Date().getFullYear()} ระบบสต๊อกร้านติดใจ (Mobile Edition)</p>
          </footer>

          {/* ALL MODALS */}
          <CategoryModal
            isOpen={categoryModalOpen}
            onClose={() => setCategoryModalOpen(false)}
            categories={categories}
            onAddCategory={handleAddCategory}
            onDeleteCategory={handleDeleteCategory}
          />

          <StockSummaryByCategoryModal
            isOpen={stockCategorySummaryModalOpen}
            onClose={() => setStockCategorySummaryModalOpen(false)}
            stocks={stocks}
            categories={categories}
          />

          <StockModal
            isOpen={stockModalOpen}
            onClose={() => setStockModalOpen(false)}
            onSave={handleSaveStock}
            editingStock={editingStock}
            handleImageUpload={handleImageUpload}
            categories={categories}
            onOpenCategoryModal={() => { setStockModalOpen(false); setCategoryModalOpen(true); }}
          />

          <RestockModal
            isOpen={restockModalOpen}
            onClose={() => setRestockModalOpen(false)}
            stocks={stocks}
            categories={categories}
            onConfirmBatchRestock={handleBatchRestock}
          />

          <TransferModal
            isOpen={transferModalOpen}
            onClose={() => setTransferModalOpen(false)}
            stocks={stocks}
            categories={categories}
            onConfirmBatchTransfer={handleBatchTransfer}
          />

          <RecipeModal
            isOpen={recipeModalOpen}
            onClose={() => setRecipeModalOpen(false)}
            onSave={handleSaveRecipe}
            editingRecipe={editingRecipe}
            stocks={stocks}
            handleImageUpload={handleImageUpload}
          />

          <EditMovementModal
            isOpen={!!movementToEdit}
            onClose={() => setMovementToEdit(null)}
            movementLog={movementToEdit}
            stocks={stocks}
            onSave={handleSaveMovementEdit}
          />

          <EditProductionLogModal
            isOpen={!!productionToEdit}
            onClose={() => setProductionToEdit(null)}
            logToEdit={productionToEdit}
            onSave={handleSaveProductionEdit}
          />

          <DeleteLogModal
            isOpen={!!logToDelete}
            onClose={() => setLogToDelete(null)}
            logToDelete={logToDelete}
            onConfirmDelete={handleConfirmDeleteLog}
          />

          <PrintSummaryModal
            isOpen={printModalOpen}
            onClose={() => setPrintModalOpen(false)}
            logs={logs}
            lowStockItems={lowStockItems}
          />
        </div>
      );
    }

    const rootElement = document.getElementById('root');
    const root = ReactDOM.createRoot(rootElement);
    root.render(<App />);
  </script>
